Nối Google Sheets của anh chị với robot

Mỗi khách tham dự để lại thông tin trên robot sẽ tự thành một dòng mới trong trang tính của anh chị, sau vài giây. Làm một lần, khoảng 10 phút, không cần cài phần mềm.

Anh chị cần: một tài khoản Google và một trang tính Google Sheets (tạo mới hoặc dùng trang có sẵn).
Xong việc, anh chị gửi lại cho Roboworld hai thứ: ① địa chỉ ứng dụng web (kết thúc bằng /exec) và ② mã bí mật anh chị tự đặt ở bước 3.

Các bước

  1. Mở trang tính Vào sheets.google.com, tạo một trang tính mới (ví dụ đặt tên “Khách sự kiện — robot”), hoặc mở trang anh chị muốn dùng. Robot sẽ tự tạo một tab tên “Khách để lại thông tin” — các tab khác không bị đụng tới.
  2. Mở Apps Script Trên thanh bảng chọn: Tiện ích mở rộng → Apps Script. Một thẻ mới mở ra với vài dòng mã có sẵn — xoá hết rồi dán đoạn mã ở cuối trang này vào.
  3. Đặt mã bí mật Ở gần đầu đoạn mã có dòng var MA_BI_MAT = 'doi-ma-nay-truoc-khi-dung';. Đổi phần trong dấu nháy thành một chuỗi riêng của anh chị, chỉ dùng chữ không dấu và số, ví dụ chibach-2026-k7m2. Bấm biểu tượng đĩa mềm để lưu.
  4. Triển khai thành ứng dụng web Góc trên bên phải: Triển khai → Tùy chọn triển khai mới → bấm bánh răng chọn loại Ứng dụng web.
    · Thực thi dưới tên: Tôi
    · Người có quyền truy cập: Bất kỳ ai
    Bấm Triển khai. Google hỏi cấp quyền → chọn tài khoản của anh chị → Nâng cao → Đi tới … (không an toàn) → Cho phép. Màn cảnh báo này hiện ra vì đoạn mã do chính anh chị dán vào, chưa qua Google thẩm định — nó chỉ có quyền ghi vào trang tính này.
  5. Sao chép địa chỉ Google hiện “URL ứng dụng web”, dạng https://script.google.com/macros/s/…/exec. Bấm Sao chép.
  6. Thử nhanh Dán địa chỉ đó vào một thẻ trình duyệt mới. Thấy dòng chữ {"ok":true, …} là ứng dụng đang chạy.
  7. Gửi cho Roboworld Gửi địa chỉ ở bước 5 và mã bí mật ở bước 3 cho kỹ thuật viên Roboworld. Bên em nạp vào robot và gửi thử một dòng — anh chị sẽ thấy dòng “DÒNG THỬ KẾT NỐI” hiện trong trang tính (xoá đi được).

Ba điều nên biết
· Mất mạng không mất khách. Robot luôn lưu vào bộ nhớ của nó trước, rồi mới gửi lên trang tính; chưa gửi được thì cứ 30 giây thử lại. Cuối sự kiện Roboworld cũng gửi anh chị một tệp Excel đối chiếu.
· Giữ kín địa chỉ và mã bí mật. Ai có cả hai là ghi được dòng vào trang tính (không đọc, không xoá được). Muốn khoá lại sau sự kiện: Triển khai → Quản lý các tùy chọn triển khai → Lưu trữ.
· Đây là dữ liệu cá nhân của khách tham dự. Anh chị chỉ chia sẻ trang tính cho người cần dùng.

Đoạn mã để dán ở bước 2

/**
 * NHẬN THÔNG TIN KHÁCH TỪ ROBOT → ghi vào trang tính này.
 *
 * Cách dùng (làm MỘT lần, bằng tài khoản Google là chủ trang tính):
 *   1. Mở trang tính → Tiện ích mở rộng → Apps Script → xoá hết mã có sẵn → dán toàn bộ tệp này.
 *   2. Đổi MA_BI_MAT bên dưới thành một chuỗi riêng của anh chị (chữ và số, không dấu).
 *   3. Triển khai → Tùy chọn triển khai mới → loại "Ứng dụng web":
 *        Thực thi dưới tên: Tôi  ·  Người có quyền truy cập: Bất kỳ ai
 *   4. Sao chép "URL ứng dụng web" (kết thúc bằng /exec) + mã bí mật, gửi cho kỹ thuật viên robot.
 *
 * Sửa mã sau này thì phải: Triển khai → Quản lý các tùy chọn triển khai → biểu tượng bút chì →
 * Phiên bản: "Phiên bản mới" → Triển khai. Không làm bước đó thì địa chỉ cũ vẫn chạy mã cũ.
 */

/** ĐỔI chuỗi này. Robot phải gửi đúng chuỗi này thì dòng mới được ghi — chặn người lạ có địa chỉ mà ghi bừa. */
var MA_BI_MAT = 'doi-ma-nay-truoc-khi-dung';

/** Tên trang (tab) sẽ chứa dữ liệu — chưa có thì tự tạo. */
var TEN_TRANG = 'Khách để lại thông tin';

var TIEU_DE = ['Thời gian', 'Họ và tên', 'Số điện thoại', 'Chức vụ', 'Doanh nghiệp / đơn vị',
               'Nguồn (chế độ trên robot)', 'Câu hỏi đang xem', 'Robot', 'Mã dòng'];

function doPost(e) {
  var khoa = LockService.getScriptLock();
  try {
    khoa.waitLock(20000);                       // hai robot gửi cùng lúc thì xếp hàng, không ghi đè nhau
    var d = JSON.parse(e.postData.contents);
    if (d.ma !== MA_BI_MAT) return tra({ ok: false, loi: 'sai ma bi mat' });

    var bang = SpreadsheetApp.getActiveSpreadsheet();
    var trang = bang.getSheetByName(TEN_TRANG) || bang.insertSheet(TEN_TRANG);
    if (trang.getLastRow() === 0) {
      trang.appendRow(TIEU_DE);
      trang.getRange(1, 1, 1, TIEU_DE.length).setFontWeight('bold').setBackground('#006097').setFontColor('#ffffff');
      trang.setFrozenRows(1);
      trang.getRange('C:C').setNumberFormat('@');   // cột số điện thoại là CHỮ — không thì mất số 0 đầu
    }

    // Robot gửi lại khi mạng chập chờn — dòng nào đã có "Mã dòng" này rồi thì bỏ qua, không ghi trùng.
    var cuoi = trang.getLastRow();
    if (d.id && cuoi > 1) {
      var tu = Math.max(2, cuoi - 300);
      var ma = trang.getRange(tu, TIEU_DE.length, cuoi - tu + 1, 1).getValues();
      for (var i = 0; i < ma.length; i++) if (ma[i][0] === d.id) return tra({ ok: true, trung: true });
    }

    trang.appendRow([d.thoi_gian || '', d.ho_ten || '', "'" + (d.so_dien_thoai || ''), d.chuc_vu || '',
                     d.don_vi || '', d.nguon || '', d.quan_tam || '', d.thiet_bi || '', d.id || '']);
    return tra({ ok: true });
  } catch (loi) {
    return tra({ ok: false, loi: String(loi) });
  } finally {
    try { khoa.releaseLock(); } catch (x) {}
  }
}

/** Mở địa chỉ /exec bằng trình duyệt để thử: thấy dòng chữ dưới đây là bản triển khai đang sống. */
function doGet() {
  return tra({ ok: true, tin: 'Dia chi nhan thong tin khach tu robot dang hoat dong.' });
}

function tra(o) {
  return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);
}
